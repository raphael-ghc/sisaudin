// COMPONENTE: SELECT COM BUSCA (COMBOBOX)
// ==========================================
function SearchableSelect({ options, value, onChange, placeholder, label, required }) {
  const [isOpen, setIsOpen] = useState(false);
  const [search, setSearch] = useState('');

  const selectedOption = options.find(opt => opt.id === value);
  const filteredOptions = options.filter(opt => 
    opt.label.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <div className="relative">
      {label && (
        <label className="block text-sm font-medium text-slate-700 mb-2">
          {label} {required && '*'}
        </label>
      )}
      <div
        className="w-full p-3 border border-slate-300 rounded-lg cursor-pointer bg-white flex justify-between items-center transition-all focus-within:ring-2 focus-within:ring-blue-500"
        onClick={() => setIsOpen(true)}
      >
        <span className={selectedOption ? 'text-slate-800' : 'text-slate-400'}>
          {selectedOption ? selectedOption.label : placeholder}
        </span>
        <ChevronDown size={16} className="text-slate-400" />
      </div>

      {isOpen && (
        <>
          <div className="fixed inset-0 z-40" onClick={() => setIsOpen(false)} />
          <div className="absolute z-50 w-full mt-1 bg-white border border-slate-300 rounded-lg shadow-xl max-h-60 flex flex-col overflow-hidden">
            <div className="p-2 border-b border-slate-100 bg-slate-50">
              <input
                type="text"
                className="w-full p-2 border border-slate-200 rounded outline-none text-sm focus:ring-2 focus:ring-blue-500"
                placeholder="Digite para buscar..."
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                autoFocus
              />
            </div>
            <div className="overflow-y-auto flex-1">
              {filteredOptions.length === 0 ? (
                <div className="p-3 text-sm text-slate-500 text-center">Nenhum resultado encontrado.</div>
              ) : (
                filteredOptions.map(opt => (
                  <div
                    key={opt.id}
                    className={`p-3 text-sm cursor-pointer hover:bg-blue-50 transition-colors ${value === opt.id ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700'}`}
                    onClick={() => {
                      onChange(opt.id);
                      setIsOpen(false);
                      setSearch('');
                    }}
                  >
                    {opt.label}
                  </div>
                ))
              )}
            </div>
          </div>
        </>
      )}
    </div>
  );
}

// COMPONENTE: SELECT MÚLTIPLO COM BUSCA
// ==========================================
function MultiSearchableSelect({ options, values = [], onChange, placeholder, label, required }) {
  const [isOpen, setIsOpen] = useState(false);
  const [search, setSearch] = useState('');

  const selectedOptions = options.filter(opt => values.includes(opt.id));
  const filteredOptions = options.filter(opt => 
    opt.label.toLowerCase().includes(search.toLowerCase())
  );

  const toggleOption = (id, e) => {
    if (e) {
      e.stopPropagation();
      e.preventDefault();
    }
    if (values.includes(id)) {
      onChange(values.filter(v => v !== id));
    } else {
      onChange([...values, id]);
    }
  };

  return (
    <div className="relative">
      {label && (
        <label className="block text-sm font-medium text-slate-700 mb-2">
          {label} {required && '*'}
        </label>
      )}
      <div
        className="w-full min-h-[46px] p-2 border border-slate-300 rounded-lg cursor-pointer bg-white flex flex-wrap gap-2 items-center transition-all focus-within:ring-2 focus-within:ring-blue-500"
        onClick={() => setIsOpen(true)}
      >
        {selectedOptions.length === 0 ? (
          <span className="text-slate-400 p-1">{placeholder}</span>
        ) : (
          selectedOptions.map(opt => (
            <span key={opt.id} className="bg-blue-100 text-blue-800 px-2 py-1 rounded text-sm flex items-center gap-1 z-10" onClick={(e) => toggleOption(opt.id, e)}>
              {opt.label} <X size={14} className="hover:text-red-500 cursor-pointer" />
            </span>
          ))
        )}
        <div className="ml-auto p-1"><ChevronDown size={16} className="text-slate-400" /></div>
      </div>

      {isOpen && (
        <>
          <div className="fixed inset-0 z-40" onClick={() => setIsOpen(false)} />
          <div className="absolute z-50 w-full mt-1 bg-white border border-slate-300 rounded-lg shadow-xl max-h-60 flex flex-col overflow-hidden">
            <div className="p-2 border-b border-slate-100 bg-slate-50">
              <input
                type="text"
                className="w-full p-2 border border-slate-200 rounded outline-none text-sm focus:ring-2 focus:ring-blue-500"
                placeholder="Digite para buscar..."
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                autoFocus
              />
            </div>
            <div className="overflow-y-auto flex-1">
              {filteredOptions.length === 0 ? (
                <div className="p-3 text-sm text-slate-500 text-center">Nenhum resultado encontrado.</div>
              ) : (
                filteredOptions.map(opt => {
                  const isSelected = values.includes(opt.id);
                  return (
                    <div
                      key={opt.id}
                      className={`p-3 text-sm cursor-pointer hover:bg-blue-50 transition-colors flex items-center justify-between ${isSelected ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700'}`}
                      onClick={() => toggleOption(opt.id)}
                    >
                      {opt.label}
                      {isSelected && <CheckSquare size={16} className="text-blue-600" />}
                    </div>
                  );
                })
              )}
            </div>
          </div>
        </>
      )}
    </div>
  );
}

// ==========================================
// MÓDULO 1: CADASTRO DE UNIDADE
// ==========================================
function CadastroUnidade({ unidades, setUnidades }) {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState({ text: '', type: '' });
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });

  const [formData, setFormData] = useState({ nome: '', sigla: '', cnpj: '', cep: '', endereco: '', uf: 'RS' });

  const showMessage = (text, type = 'success') => {
    setMessage({ text, type });
    setTimeout(() => setMessage({ text: '', type: '' }), 3000);
  };

  const handleOpenForm = (unidade = null) => {
    if (unidade) { setFormData(unidade); setEditingId(unidade.id); } 
    else { setFormData({ nome: '', sigla: '', cnpj: '', cep: '', endereco: '', uf: 'RS' }); setEditingId(null); }
    setIsFormOpen(true);
    scrollToTop();
  };

  const handleCloseForm = () => { setIsFormOpen(false); setEditingId(null); };

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    let formattedValue = value;
    if (name === 'cnpj') formattedValue = value.replace(/\D/g, '').replace(/^(\d{2})(\d)/, '$1.$2').replace(/^(\d{2})\.(\d{3})(\d)/, '$1.$2.$3').replace(/\.(\d{3})(\d)/, '.$1/$2').replace(/(\d{4})(\d)/, '$1-$2').substring(0, 18);
    else if (name === 'cep') formattedValue = value.replace(/\D/g, '').replace(/^(\d{5})(\d)/, '$1-$2').substring(0, 9);
    if (name === 'sigla') formattedValue = formattedValue.toUpperCase();
    setFormData({ ...formData, [name]: formattedValue });
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.nome || !formData.sigla || !formData.cnpj) return showMessage('Preencha todos os campos obrigatórios.', 'error');

    if (editingId) {
      setUnidades(unidades.map(u => u.id === editingId ? { ...formData, id: editingId } : u));
      showMessage('Unidade atualizada com sucesso!');
    } else {
      setUnidades([...unidades, { ...formData, id: Date.now() }]);
      showMessage('Nova unidade cadastrada com sucesso!');
    }
    handleCloseForm();
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setUnidades(unidades.filter(u => u.id !== deleteModal.data.id));
      showMessage('Unidade excluída com sucesso!');
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  return (
    <div className="max-w-6xl mx-auto space-y-6">
      <HeaderSection title="Cadastro de Unidades" onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

      {isFormOpen && (
        <div className="bg-white p-6 rounded-xl shadow-md border border-slate-200 animate-in fade-in slide-in-from-top-4">
          <FormHeader title={editingId ? 'Editar Unidade' : 'Cadastrar Nova Unidade'} onClose={handleCloseForm} />
          <form onSubmit={handleSubmit}>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-6">
              <div className="md:col-span-2">
                <label className="block text-sm font-medium text-slate-700 mb-2">Nome da Unidade *</label>
                <input type="text" name="nome" value={formData.nome} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Ex: Hospital Nossa Senhora da Conceição" required autoFocus />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Sigla *</label>
                <input type="text" name="sigla" value={formData.sigla} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none uppercase" placeholder="Ex: HNSC" maxLength="10" required />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">CNPJ *</label>
                <input type="text" name="cnpj" value={formData.cnpj} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="00.000.000/0000-00" maxLength="18" required />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">CEP</label>
                <input type="text" name="cep" value={formData.cep} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="00000-000" maxLength="9" />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Estado (UF) *</label>
                <select name="uf" value={formData.uf} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none bg-white">
                  {ESTADOS_BR.map(uf => <option key={uf} value={uf}>{uf}</option>)}
                </select>
              </div>
              <div className="md:col-span-2">
                <label className="block text-sm font-medium text-slate-700 mb-2">Endereço Completo</label>
                <input type="text" name="endereco" value={formData.endereco} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Rua, Número, Bairro" />
              </div>
            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />
          </form>
        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Sigla</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Nome da Unidade</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">CNPJ</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Ações</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {unidades.length === 0 ? <EmptyRow cols={4} /> : unidades.map((u) => (
              <tr key={u.id} className="hover:bg-slate-50/50 group">
                <td className="px-6 py-4 text-sm font-bold text-blue-900">{u.sigla}</td>
                <td className="px-6 py-4 text-sm font-medium text-slate-800">{u.nome}</td>
                <td className="px-6 py-4 text-sm text-slate-600">{u.cnpj}</td>
                <td className="px-6 py-4 text-sm text-center">
                  <ActionButtons onEdit={() => handleOpenForm(u)} onDelete={() => setDeleteModal({ isOpen: true, data: u })} />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.nome} />
    </div>
  );
}

function CadastroCargos({ cargos, setCargos }) {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState({ text: '', type: '' });
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });

  const [formData, setFormData] = useState({ nome: '' });

  const showMessage = (text, type = 'success') => {
    setMessage({ text, type });
    setTimeout(() => setMessage({ text: '', type: '' }), 3000);
  };

  const handleOpenForm = (cargo = null) => {
    if (cargo) { setFormData(cargo); setEditingId(cargo.id); } 
    else { setFormData({ nome: '' }); setEditingId(null); }
    setIsFormOpen(true);
    scrollToTop();
  };

  const handleCloseForm = () => { setIsFormOpen(false); setEditingId(null); };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.nome) return showMessage('O nome do cargo é obrigatório.', 'error');

      setCargos(cargos.map(c => c.id === editingId ? { ...formData, id: editingId } : c));

    } else {
      setCargos([...cargos, { ...formData, id: Date.now() }]);

      setCargos(cargos.filter(c => c.id !== deleteModal.data.id));
      showMessage('Cargo excluído com sucesso!');
      setDeleteModal({ isOpen: false, data: null });

  return (
    <div className="max-w-4xl mx-auto space-y-6">
      <HeaderSection title="Cadastro de Cargos" onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

          <FormHeader title={editingId ? 'Editar Cargo' : 'Cadastrar Novo Cargo'} onClose={handleCloseForm} />
          <form onSubmit={handleSubmit}>
            <div className="mb-6">
              <label className="block text-sm font-medium text-slate-700 mb-2">Nome do Cargo *</label>
              <input type="text" value={formData.nome} onChange={(e) => setFormData({nome: e.target.value})} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Ex: Auditor Sênior" required autoFocus />
            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />

        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">ID</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 w-full">Nome do Cargo</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Ações</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {cargos.length === 0 ? <EmptyRow cols={3} /> : cargos.map((c) => (
              <tr key={c.id} className="hover:bg-slate-50/50 group">

                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.nome} />

    const isCartaoDup = pessoas.some(p => p.cartaoPonto === formData.cartaoPonto && p.id !== editingId);
    if (isCartaoDup) return showMessage('Já existe uma pessoa com este Cartão Ponto.', 'error');
    
    const isEmailDup = pessoas.some(p => p.email.toLowerCase() === formData.email.toLowerCase() && p.id !== editingId);
    if (isEmailDup) return showMessage('Já existe uma pessoa com este E-mail.', 'error');

    if (editingId) {
      setPessoas(pessoas.map(p => p.id === editingId ? { ...formData, id: editingId } : p));
      showMessage('Pessoa atualizada com sucesso!');
    } else {
      setPessoas([...pessoas, { ...formData, id: Date.now() }]);
      showMessage('Nova pessoa cadastrada com sucesso!');
    }
    handleCloseForm();
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setPessoas(pessoas.filter(p => p.id !== deleteModal.data.id));
      showMessage('Pessoa excluída com sucesso!');
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  return (
    <div className="max-w-6xl mx-auto space-y-6">
      <HeaderSection title="Cadastro de Pessoas" onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

      {isFormOpen && (
        <div className="bg-white p-6 rounded-xl shadow-md border border-slate-200 animate-in fade-in">
          <FormHeader title={editingId ? 'Editar Pessoa' : 'Cadastrar Nova Pessoa'} onClose={handleCloseForm} />
          <form onSubmit={handleSubmit}>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-6">
              
              <div className="md:col-span-2">
                <label className="block text-sm font-medium text-slate-700 mb-2">Nome Completo *</label>
                <input type="text" name="nome" value={formData.nome} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Nome da pessoa" required autoFocus />
              </div>

              <div>
                <SearchableSelect 
                  label="Unidade Institucional" required 
                  options={unidadeOptions} 
                  value={formData.unidadeId} 
                  onChange={(val) => setFormData({...formData, unidadeId: val})} 
                  placeholder="Busque e selecione..." 
                />
              </div>

              <div>
                <SearchableSelect 
                  label="Gerência" required 
                  options={gerenciaOptions} 
                  value={formData.gerenciaId} 
                  onChange={(val) => setFormData({...formData, gerenciaId: val})} 
                  placeholder="Busque e selecione..." 
                />
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Cargo *</label>
                <select name="cargoId" value={formData.cargoId} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none bg-white" required>
                  <option value="">Selecione um Cargo</option>
                  {cargos.map(c => <option key={c.id} value={c.id}>{c.nome}</option>)}
                </select>
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Cartão Ponto *</label>
                <input type="text" name="cartaoPonto" value={formData.cartaoPonto} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Ex: C123456" required />
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">E-mail *</label>
                <input type="email" name="email" value={formData.email} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="email@ghc.com.br" required />
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Ramal *</label>
                <input type="text" name="ramal" value={formData.ramal} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Ex: 1234" required />
              </div>

            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />
          </form>
        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Nome / E-mail</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Cartão / Ramal</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Alocação</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Ações</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {pessoas.length === 0 ? <EmptyRow cols={4} /> : pessoas.map((p) => {
              const unidade = unidades.find(u => u.id === p.unidadeId);
              const gerencia = gerencias.find(g => g.id === p.gerenciaId);
              const cargo = cargos.find(c => c.id === p.cargoId);
              return (
                <tr key={p.id} className="hover:bg-slate-50/50 group">
                  <td className="px-6 py-4">
                    <div className="text-sm font-medium text-slate-800">{p.nome}</div>
                    <div className="text-xs text-slate-500">{p.email}</div>
                  </td>

                    <div>Ponto: <span className="font-medium text-slate-800">{p.cartaoPonto}</span></div>
                    <div>Ramal: <span className="font-medium text-slate-800">{p.ramal}</span></div>
                  </td>
                  <td className="px-6 py-4 text-sm text-slate-600">
                    <div>{cargo?.nome || '-'}</div>
                    <div className="text-xs text-blue-600">{gerencia?.sigla} ({unidade?.sigla})</div>
                  </td>
                  <td className="px-6 py-4 text-sm text-center">
                    <ActionButtons onEdit={() => handleOpenForm(p)} onDelete={() => setDeleteModal({ isOpen: true, data: p })} />
                  </td>
                </tr>
              )
            })}
          </tbody>
        </table>
      </div>
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.nome} />
    </div>
  );
}

// MÓDULO: CADASTRO DE USUÁRIOS DO SISTEMA
// ==========================================
function CadastroUsuarios({ usuarios, setUsuarios }) {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState({ text: '', type: '' });
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });

  const [formData, setFormData] = useState({ nome: '', email: '', foto: null });

  const showMessage = (text, type = 'success') => {
    setMessage({ text, type });
    setTimeout(() => setMessage({ text: '', type: '' }), 3000);
  };

  const handleOpenForm = (usuario = null) => {
    if (usuario) { setFormData(usuario); setEditingId(usuario.id); } 
    else { setFormData({ nome: '', email: '', foto: null }); setEditingId(null); }
    setIsFormOpen(true);
    scrollToTop();
  };

  const handleCloseForm = () => { setIsFormOpen(false); setEditingId(null); };

  const handleInputChange = (e) => {
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  const handleFileChange = (e) => {
    const file = e.target.files[0];
    if (file) {
      if (!file.type.startsWith('image/')) {
        return showMessage('Por favor, selecione um arquivo de imagem válido.', 'error');
      }
      const reader = new FileReader();
      reader.onloadend = () => {
        setFormData({ ...formData, foto: reader.result });
      };
      reader.readAsDataURL(file);
    }
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.nome || !formData.email) return showMessage('Preencha os campos obrigatórios.', 'error');

    const isEmailDup = usuarios.some(u => u.email.toLowerCase() === formData.email.toLowerCase() && u.id !== editingId);
    if (isEmailDup) return showMessage('Já existe um usuário com este e-mail.', 'error');

    if (editingId) {
      setUsuarios(usuarios.map(u => u.id === editingId ? { ...formData, id: editingId } : u));
      showMessage('Usuário atualizado com sucesso!');
    } else {
      setUsuarios([...usuarios, { ...formData, id: Date.now() }]);
      showMessage('Novo usuário cadastrado com sucesso!');
    }
    handleCloseForm();
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setUsuarios(usuarios.filter(u => u.id !== deleteModal.data.id));
      showMessage('Usuário excluído com sucesso!');
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  return (
    <div className="max-w-4xl mx-auto space-y-6 animate-in fade-in">
      <HeaderSection title="Cadastro de Usuários" onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

      {isFormOpen && (
        <div className="bg-white p-6 rounded-xl shadow-md border border-slate-200">
          <FormHeader title={editingId ? 'Editar Usuário' : 'Cadastrar Novo Usuário'} onClose={handleCloseForm} />
          <form onSubmit={handleSubmit}>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-6">
              
              <div className="md:col-span-2 flex items-center gap-6 p-4 bg-slate-50 border border-slate-200 rounded-xl">
                <div className="relative shrink-0 w-24 h-24 rounded-full border-4 border-white shadow-md overflow-hidden bg-slate-200 flex items-center justify-center">
                  {formData.foto ? (
                    <img src={formData.foto} alt="Preview da Foto" className="w-full h-full object-cover" />
                  ) : (
                    <User size={40} className="text-slate-400" />
                  )}
                </div>
                <div className="flex-1">
                  <label className="block text-sm font-medium text-slate-700 mb-2">Foto de Perfil</label>
                  <label className="cursor-pointer inline-flex items-center gap-2 px-4 py-2 bg-white border border-slate-300 rounded-lg shadow-sm text-sm font-medium text-slate-700 hover:bg-slate-50 transition-colors">
                    <Camera size={16} className="text-slate-500" />
                    Escolher imagem
                    <input type="file" accept="image/*" onChange={handleFileChange} className="hidden" />
                  </label>
                  <p className="text-xs text-slate-500 mt-2">Formatos suportados: JPG, PNG, GIF. Máximo 2MB recomendado.</p>
                </div>
              </div>

              <div className="md:col-span-1">
                <label className="block text-sm font-medium text-slate-700 mb-2">Nome Completo *</label>
                <input type="text" name="nome" value={formData.nome} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Ex: João Silva" required autoFocus />
              </div>

              <div className="md:col-span-1">
                <label className="block text-sm font-medium text-slate-700 mb-2">E-mail de Acesso *</label>
                <input type="email" name="email" value={formData.email} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="joao@ghc.com.br" required />
              </div>

            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />
          </form>
        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 w-20">Foto</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Nome do Usuário</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">E-mail</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Ações</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {usuarios.length === 0 ? <EmptyRow cols={4} /> : usuarios.map((u) => (
              <tr key={u.id} className="hover:bg-slate-50/50 group">
                <td className="px-6 py-4 text-sm">
                  <div className="w-10 h-10 rounded-full bg-slate-200 border border-slate-300 flex items-center justify-center overflow-hidden">
                    {u.foto ? (
                      <img src={u.foto} alt={u.nome} className="w-full h-full object-cover" />
                    ) : (
                      <span className="font-bold text-slate-500 text-xs">
                        {u.nome.substring(0, 2).toUpperCase()}
                      </span>
                    )}
                  </div>
                </td>
                <td className="px-6 py-4 text-sm font-medium text-slate-800">{u.nome}</td>
                <td className="px-6 py-4 text-sm text-slate-600">{u.email}</td>
                <td className="px-6 py-4 text-sm text-center">
                  <ActionButtons onEdit={() => handleOpenForm(u)} onDelete={() => setDeleteModal({ isOpen: true, data: u })} />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.nome} />
    </div>
  );
}

// MÓDULO 5: CADASTROS DE DOMÍNIO (GENÉRICO)
// ==========================================
function CadastroDominio({ title, data, setData, hasAtivo = false }) {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState({ text: '', type: '' });
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });

  const [formData, setFormData] = useState({ nome: '', ativo: false });

  const showMessage = (text, type = 'success') => {
    setMessage({ text, type });
    setTimeout(() => setMessage({ text: '', type: '' }), 3000);
  };

  const handleOpenForm = (item = null) => {
    if (item) { setFormData({ nome: item.nome, ativo: item.ativo || false }); setEditingId(item.id); } 
    else { setFormData({ nome: '', ativo: false }); setEditingId(null); }
    setIsFormOpen(true);
    scrollToTop();
  };

  const handleCloseForm = () => { setIsFormOpen(false); setEditingId(null); };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.nome) return showMessage('O nome é obrigatório.', 'error');
    
    const isDuplicate = data.some(d => d.nome.toLowerCase() === formData.nome.toLowerCase() && d.id !== editingId);
    if (isDuplicate) return showMessage(`Já existe um registro com este nome.`, 'error');

    if (editingId) {
      setData(data.map(d => d.id === editingId ? { ...formData, id: editingId } : d));
      showMessage('Registro atualizado com sucesso!');
    } else {
      setData([...data, { ...formData, id: Date.now() }]);
      showMessage('Novo registro cadastrado com sucesso!');
    }
    handleCloseForm();
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setData(data.filter(d => d.id !== deleteModal.data.id));
      showMessage('Registro excluído com sucesso!');
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  return (
    <div className="max-w-4xl mx-auto space-y-6 animate-in fade-in">
      <HeaderSection title={title} onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

      {isFormOpen && (
        <div className="bg-white p-6 rounded-xl shadow-md border border-slate-200">
          <FormHeader title={editingId ? `Editar ${title}` : `Cadastrar ${title}`} onClose={handleCloseForm} />
          <form onSubmit={handleSubmit}>
            <div className="mb-6 space-y-4">
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Nome *</label>
                <input type="text" value={formData.nome} onChange={(e) => setFormData({...formData, nome: e.target.value})} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Digite o nome" required autoFocus />
              </div>
              {hasAtivo && (
                <div className="flex items-center gap-2 mt-4 bg-slate-50 p-3 rounded-lg border border-slate-200">
                  <input type="checkbox" id="ativoCheckbox" checked={formData.ativo} onChange={(e) => setFormData({...formData, ativo: e.target.checked})} className="w-4 h-4 text-blue-600 rounded border-gray-300 focus:ring-blue-500" />
                  <label htmlFor="ativoCheckbox" className="text-sm font-medium text-slate-700 cursor-pointer">Registro Ativo?</label>
                </div>
              )}
            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />
          </form>
        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 w-16">ID</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Nome do Registro</th>
              {hasAtivo && <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Status</th>}
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center w-32">Ações</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {data.length === 0 ? <EmptyRow cols={hasAtivo ? 4 : 3} /> : data.map((d) => (
              <tr key={d.id} className="hover:bg-slate-50/50 group">
                <td className="px-6 py-4 text-sm text-slate-500">#{d.id}</td>
                <td className="px-6 py-4 text-sm font-medium text-slate-800">{d.nome}</td>
                {hasAtivo && (
                  <td className="px-6 py-4 text-sm text-center">
                    <span className={`py-1 px-2 rounded font-medium text-xs ${d.ativo ? 'bg-emerald-100 text-emerald-700' : 'bg-slate-100 text-slate-600'}`}>
                      {d.ativo ? 'Ativo' : 'Inativo'}
                    </span>
                  </td>
                )}
                <td className="px-6 py-4 text-sm text-center">
                  <ActionButtons onEdit={() => handleOpenForm(d)} onDelete={() => setDeleteModal({ isOpen: true, data: d })} />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.nome} />
    </div>
  );
}

// MÓDULO 6: CADASTRO DE ITENS DE TRABALHO
// ==========================================
function CadastroItensTrabalho({ itens, setItens, tiposItem, tiposServico, macroprocessos, origensDemanda, niveisRisco }) {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState({ text: '', type: '' });
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });

  const [formData, setFormData] = useState({
    tipoItemTrabalhoId: '', tipoServicoId: '', macroprocessoId: '', anoReferencia: '', numeroReferencia: '',
    dataInicial: '', dataFinal: '', qtdHora: '', origemDemandaId: '', nivelRiscoId: '',
    objetoAuditado: '', objetivoAuditoria: '', objetoAuditadoPaint: '', objetivoAuditoriaPaint: ''
  });

  const tiposItemOptions = tiposItem.map(t => ({ id: t.id, label: t.nome }));
  const tiposServicoOptions = tiposServico.map(t => ({ id: t.id, label: t.nome }));
  const macroprocessoOptions = macroprocessos.map(m => ({ id: m.id, label: m.nome }));
  const origensDemandaOptions = origensDemanda.map(o => ({ id: o.id, label: o.nome }));
  const niveisRiscoOptions = niveisRisco.map(n => ({ id: n.id, label: n.nome }));

  const tipoItemSelecionado = tiposItem.find(t => t.id === formData.tipoItemTrabalhoId);
  const isPaint = tipoItemSelecionado?.nome.toUpperCase() === 'PAINT';

  const showMessage = (text, type = 'success') => {
    setMessage({ text, type });
    setTimeout(() => setMessage({ text: '', type: '' }), 3000);
  };

  const handleOpenForm = (item = null) => {
    if (item) { setFormData(item); setEditingId(item.id); } 
    else { 
      setFormData({
        tipoItemTrabalhoId: '', tipoServicoId: '', macroprocessoId: '', anoReferencia: new Date().getFullYear().toString(), 
        numeroReferencia: '', dataInicial: '', dataFinal: '', qtdHora: '', origemDemandaId: '', nivelRiscoId: '',
        objetoAuditado: '', objetivoAuditoria: '', objetoAuditadoPaint: '', objetivoAuditoriaPaint: ''
      }); 
      setEditingId(null); 
    }
    setIsFormOpen(true);
    scrollToTop();
  };

  const handleCloseForm = () => { setIsFormOpen(false); setEditingId(null); };
  const handleInputChange = (e) => setFormData({ ...formData, [e.target.name]: e.target.value });

  const handleSubmit = (e) => {
    e.preventDefault();
    
    if (!formData.anoReferencia || !formData.numeroReferencia || !formData.dataInicial || !formData.dataFinal || !formData.qtdHora || !formData.objetoAuditado || !formData.objetivoAuditoria) {
      return showMessage('Preencha todos os campos obrigatórios textuais.', 'error');
    }
    
    if (!formData.tipoItemTrabalhoId || !formData.tipoServicoId || !formData.macroprocessoId || !formData.origemDemandaId || !formData.nivelRiscoId) {
      return showMessage('Preencha todos os campos de seleção obrigatórios.', 'error');
    }

    if (isPaint && (!formData.objetoAuditadoPaint || !formData.objetivoAuditoriaPaint)) {
      return showMessage('Preencha os campos obrigatórios do PAINT.', 'error');
    }

    if (editingId) {
      setItens(itens.map(i => i.id === editingId ? { ...formData, id: editingId } : i));
      showMessage('Item de trabalho atualizado com sucesso!');
    } else {
      setItens([...itens, { ...formData, id: Date.now() }]);
      showMessage('Novo item de trabalho cadastrado com sucesso!');
    }
    handleCloseForm();
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setItens(itens.filter(i => i.id !== deleteModal.data.id));
      showMessage('Item de Trabalho excluído com sucesso!');
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  return (
    <div className="max-w-7xl mx-auto space-y-6 animate-in fade-in">
      <HeaderSection title="Itens de Trabalho" onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

      {isFormOpen && (
        <div className="bg-white p-6 rounded-xl shadow-md border border-slate-200">
          <FormHeader title={editingId ? 'Editar Item de Trabalho' : 'Cadastrar Novo Item'} onClose={handleCloseForm} />
          <form onSubmit={handleSubmit}>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mb-6">
              
              <div className="md:col-span-1">
                <SearchableSelect label="Tipo de Item de Trabalho" required options={tiposItemOptions} value={formData.tipoItemTrabalhoId} onChange={(val) => setFormData({...formData, tipoItemTrabalhoId: val})} placeholder="Selecione o Tipo" />
              </div>
              <div className="md:col-span-1">
                <SearchableSelect label="Tipo de Serviço" required options={tiposServicoOptions} value={formData.tipoServicoId} onChange={(val) => setFormData({...formData, tipoServicoId: val})} placeholder="Selecione o Serviço" />
              </div>
              <div className="md:col-span-1">
                <SearchableSelect label="Macroprocesso" required options={macroprocessoOptions} value={formData.macroprocessoId} onChange={(val) => setFormData({...formData, macroprocessoId: val})} placeholder="Selecione o Macroprocesso" />
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Ano de Referência *</label>
                <input type="number" name="anoReferencia" value={formData.anoReferencia} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" required />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Número de Referência *</label>
                <input type="text" name="numeroReferencia" value={formData.numeroReferencia} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" required />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Quantidade de Horas *</label>
                <input type="number" name="qtdHora" value={formData.qtdHora} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" required />
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Data Inicial Prevista *</label>
                <input type="date" name="dataInicial" value={formData.dataInicial} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" required />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-2">Data Final Prevista *</label>
                <input type="date" name="dataFinal" value={formData.dataFinal} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" required />
              </div>

                <SearchableSelect label="Origem da Demanda" required options={origensDemandaOptions} value={formData.origemDemandaId} onChange={(val) => setFormData({...formData, origemDemandaId: val})} placeholder="Selecione a Origem" />

                <SearchableSelect label="Nível de Risco" required options={niveisRiscoOptions} value={formData.nivelRiscoId} onChange={(val) => setFormData({...formData, nivelRiscoId: val})} placeholder="Selecione o Risco" />
              </div>

              {isPaint && (
                <>
                  <div className="md:col-span-3 bg-blue-50 p-4 rounded-lg border border-blue-100">
                    <h4 className="text-sm font-bold text-blue-800 mb-4 flex items-center gap-2"><CheckSquare size={16} /> Campos Específicos PAINT</h4>
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                      <div>
                        <label className="block text-sm font-medium text-slate-700 mb-2">Objeto Auditado PAINT *</label>
                        <textarea name="objetoAuditadoPaint" value={formData.objetoAuditadoPaint} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" rows="3" required={isPaint}></textarea>
                      </div>
                      <div>
                        <label className="block text-sm font-medium text-slate-700 mb-2">Objetivo da Auditoria PAINT *</label>
                        <textarea name="objetivoAuditoriaPaint" value={formData.objetivoAuditoriaPaint} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" rows="3" required={isPaint}></textarea>
                      </div>
                    </div>
                  </div>
                </>
              )}

              <div className="md:col-span-3 border-t pt-4 mt-2">
                <h4 className="text-sm font-bold text-slate-800 mb-4">Informações Gerais do Objeto</h4>
                <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                  <div>
                    <label className="block text-sm font-medium text-slate-700 mb-2">Objeto Auditado *</label>
                    <textarea name="objetoAuditado" value={formData.objetoAuditado} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" rows="3" required></textarea>
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-slate-700 mb-2">Objetivo da Auditoria *</label>
                    <textarea name="objetivoAuditoria" value={formData.objetivoAuditoria} onChange={handleInputChange} className="w-full p-3 border border-slate-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" rows="3" required></textarea>
                  </div>
                </div>
              </div>

            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />
          </form>
        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse whitespace-nowrap">
            <thead>
              <tr className="bg-slate-50 border-b border-slate-200">
                <th className="px-6 py-4 text-sm font-semibold text-slate-600">Referência</th>
                <th className="px-6 py-4 text-sm font-semibold text-slate-600">Tipo</th>
                <th className="px-6 py-4 text-sm font-semibold text-slate-600">Serviço / Macroprocesso</th>
                <th className="px-6 py-4 text-sm font-semibold text-slate-600">Período Previsto</th>
                <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Ações</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-100">
              {itens.length === 0 ? <EmptyRow cols={5} /> : itens.map((i) => {
                const tipoItem = tiposItem.find(t => t.id === i.tipoItemTrabalhoId);
                const tipoServ = tiposServico.find(t => t.id === i.tipoServicoId);
                const macro = macroprocessos.find(m => m.id === i.macroprocessoId);
                
                return (
                  <tr key={i.id} className="hover:bg-slate-50/50 group">
                    <td className="px-6 py-4">
                      <div className="text-sm font-bold text-slate-800">{i.numeroReferencia}</div>
                      <div className="text-xs text-slate-500">Ano: {i.anoReferencia}</div>
                    </td>
                    <td className="px-6 py-4 text-sm font-medium text-blue-700">{tipoItem?.nome || '-'}</td>
                    <td className="px-6 py-4 text-sm text-slate-600">
                      <div>{tipoServ?.nome || '-'}</div>
                      <div className="text-xs text-slate-400">{macro?.nome || '-'}</div>
                    </td>
                    <td className="px-6 py-4 text-sm text-slate-600">
                      <div>De: {new Date(i.dataInicial).toLocaleDateString('pt-BR')}</div>
                      <div>Até: {new Date(i.dataFinal).toLocaleDateString('pt-BR')}</div>
                    </td>
                    <td className="px-6 py-4 text-sm text-center">
                      <ActionButtons onEdit={() => handleOpenForm(i)} onDelete={() => setDeleteModal({ isOpen: true, data: i })} />
                    </td>
                  </tr>
                )
              })}
            </tbody>
          </table>
        </div>
      </div>
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.numeroReferencia} />
    </div>
  );
}


// Componentes Auxiliares
function HeaderSection({ title, onNew, isFormOpen }) {
  return (
    <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 bg-white p-6 rounded-xl shadow-sm border border-slate-200">
      <div>
        <h1 className="text-2xl font-bold text-slate-800">{title}</h1>
      </div>
      {!isFormOpen && onNew && (
        <button onClick={onNew} className="bg-blue-700 hover:bg-blue-800 text-white px-5 py-2.5 rounded-lg flex items-center justify-center gap-2 font-medium transition-colors shadow-sm">
          <Plus size={20} /> Novo Registro
        </button>
      )}
    </div>
  );
}
function FeedbackMessage({ message }) {
  if (!message.text) return null;
  return (
    <div className={`p-4 rounded-lg flex items-center gap-3 shadow-sm border ${message.type === 'error' ? 'bg-red-50 text-red-700 border-red-200' : 'bg-emerald-50 text-emerald-700 border-emerald-200'}`}>
      {message.type === 'error' ? <AlertCircle size={20} /> : <ShieldCheck size={20} />}
      <span className="font-medium">{message.text}</span>
    </div>
  );
}
function FormHeader({ title, onClose }) {
  return (
    <div className="flex items-center justify-between mb-6 border-b pb-4">
      <h2 className="text-lg font-bold text-slate-800">{title}</h2>
      <button type="button" onClick={onClose} className="text-slate-400 hover:text-slate-600 bg-slate-100 hover:bg-slate-200 p-2 rounded-full transition-colors"><X size={20} /></button>
    </div>
  );
}
function FormActions({ onCancel, editingId, submitText }) {
  return (
    <div className="flex justify-end gap-3 pt-4 border-t">
      <button type="button" onClick={onCancel} className="px-6 py-2.5 border border-slate-300 text-slate-700 font-medium rounded-lg hover:bg-slate-50 transition-colors">Cancelar</button>
      <button type="submit" className="px-6 py-2.5 bg-blue-700 text-white font-medium rounded-lg hover:bg-blue-800 transition-colors flex items-center gap-2">
        <Save size={18} /> {submitText || (editingId ? 'Salvar Alterações' : 'Cadastrar')}

    </div>
  );
}
function ActionButtons({ onEdit, onDelete, children }) {
  return (
    <div className="flex items-center justify-center gap-2 opacity-80 group-hover:opacity-100 transition-opacity">
      {children}
      <button onClick={onEdit} className="p-2 text-blue-600 bg-blue-50 hover:bg-blue-100 rounded-lg transition-colors" title="Editar"><Edit2 size={16} /></button>
      <button onClick={onDelete} className="p-2 text-red-600 bg-red-50 hover:bg-red-100 rounded-lg transition-colors" title="Excluir"><Trash2 size={16} /></button>
    </div>

}
function EmptyRow({ cols, text }) {
  return <tr><td colSpan={cols} className="px-6 py-8 text-center text-slate-500">{text || 'Nenhum registro encontrado no sistema.'}</td></tr>;
}

  return (
    <div className="fixed inset-0 bg-slate-900/50 backdrop-blur-sm z-[90] flex items-center justify-center p-4">
      <div className="bg-white rounded-xl shadow-2xl max-w-md w-full p-6 animate-in fade-in zoom-in-95 duration-200">
        <div className="flex items-start gap-4 mb-4">
          <div className="bg-red-100 p-3 rounded-full text-red-600 shrink-0"><AlertTriangle size={24} /></div>
          <div>

        </div>
        <div className="flex justify-end gap-3 mt-6">

// MÓDULO: CADASTRO DE PROCESSOS
// ==========================================
function CadastroProcessos({ processos, setProcessos, itensTrabalho, pessoas, gerencias, documentos, setDocumentos, macroprocessos, origensDemanda, niveisRisco, tiposServico }) {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState({ text: '', type: '' });
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });
  const [docsModal, setDocsModal] = useState({ isOpen: false, processo: null });

  const gerenciaAuditoria = gerencias.find(g => g.sigla === 'DIRAUD' || g.id === 1);
  const supervisor = pessoas.find(p => p.id === gerenciaAuditoria?.gerenteId);
  const pessoasAuditoria = pessoas.filter(p => p.gerenciaId === gerenciaAuditoria?.id);

  const [formData, setFormData] = useState({ itemTrabalhoId: '', responsavelId: '', equipeIds: [] });

  const itemTrabalhoOptions = itensTrabalho.map(i => ({ id: i.id, label: `${i.numeroReferencia} (${i.anoReferencia}) - ${i.objetoAuditado}` }));
  const pessoaAuditoriaOptions = pessoasAuditoria.map(p => ({ id: p.id, label: `${p.nome} (Cartão: ${p.cartaoPonto})` }));

  const showMessage = (text, type = 'success') => { setMessage({ text, type }); setTimeout(() => setMessage({ text: '', type: '' }), 3000); };
  const handleOpenForm = (processo = null) => {
    if (processo) { setFormData({ itemTrabalhoId: processo.itemTrabalhoId, responsavelId: processo.responsavelId, equipeIds: processo.equipeIds || [] }); setEditingId(processo.id); } 
    else { setFormData({ itemTrabalhoId: '', responsavelId: '', equipeIds: [] }); setEditingId(null); }
    setIsFormOpen(true); scrollToTop();
  };
  const handleCloseForm = () => { setIsFormOpen(false); setEditingId(null); };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.itemTrabalhoId || !formData.responsavelId || formData.equipeIds.length === 0) return showMessage('Preencha os campos obrigatórios.', 'error');
    if (!supervisor) return showMessage('Supervisor não identificado.', 'error');
    
    const payload = { ...formData, supervisorId: supervisor.id };
    if (editingId) { setProcessos(processos.map(p => p.id === editingId ? { ...payload, id: editingId } : p)); showMessage('Processo atualizado!'); } 
    else { setProcessos([...processos, { ...payload, id: Date.now() }]); showMessage('Novo Processo cadastrado!'); }
    handleCloseForm();
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setProcessos(processos.filter(p => p.id !== deleteModal.data.id));
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  return (
    <div className="max-w-5xl mx-auto space-y-6 animate-in fade-in">
      <HeaderSection title="Cadastro de Processos" onNew={() => handleOpenForm()} isFormOpen={isFormOpen} />
      <FeedbackMessage message={message} />

      {isFormOpen && (
        <div className="bg-white p-6 rounded-xl shadow-md border border-slate-200">
          <FormHeader title={editingId ? 'Editar Processo' : 'Cadastrar Novo Processo'} onClose={handleCloseForm} />
          {!supervisor && <div className="bg-yellow-50 text-yellow-800 p-4 rounded-lg mb-6 text-sm flex gap-2"><AlertTriangle size={20}/><span>Aviso: A Gerência de Auditoria não possui Gerente.</span></div>}
          <form onSubmit={handleSubmit}>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-6">
              <div className="md:col-span-2"><SearchableSelect label="Item de Trabalho" required options={itemTrabalhoOptions} value={formData.itemTrabalhoId} onChange={(val) => setFormData({...formData, itemTrabalhoId: val})} placeholder="Selecione um Item" /></div>
              <div className="md:col-span-2">
                <label className="block text-sm font-medium text-slate-700 mb-2">Supervisor *</label>
                <div className="w-full p-3 border border-slate-200 bg-slate-100 rounded-lg text-slate-500 cursor-not-allowed">{supervisor ? supervisor.nome : 'Não identificado'}</div>
              </div>
              <div className="md:col-span-2"><SearchableSelect label="Responsável" required options={pessoaAuditoriaOptions} value={formData.responsavelId} onChange={(val) => setFormData({...formData, responsavelId: val})} placeholder="Selecione" /></div>
              <div className="md:col-span-2"><MultiSearchableSelect label="Equipe" required options={pessoaAuditoriaOptions} values={formData.equipeIds} onChange={(vals) => setFormData({...formData, equipeIds: vals})} placeholder="Selecione" /></div>
            </div>
            <FormActions onCancel={handleCloseForm} editingId={editingId} />
          </form>
        </div>
      )}

      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Item de Trabalho</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600">Responsável</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Docs</th>
              <th className="px-6 py-4 text-sm font-semibold text-slate-600 text-center">Ações</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {processos.length === 0 ? <EmptyRow cols={4} /> : processos.map((p) => {
              const item = itensTrabalho.find(i => i.id === p.itemTrabalhoId);
              const resp = pessoas.find(pes => pes.id === p.responsavelId);
              return (
                <tr key={p.id} className="hover:bg-slate-50/50 group">
                  <td className="px-6 py-4"><div className="text-sm font-bold">{item ? item.numeroReferencia : '-'}</div></td>
                  <td className="px-6 py-4"><div className="text-sm">{resp ? resp.nome : '-'}</div></td>
                  <td className="px-6 py-4 text-center">
                    <button onClick={() => setDocsModal({ isOpen: true, processo: p })} className="px-3 py-1.5 bg-emerald-50 text-emerald-700 rounded-md font-medium text-xs flex items-center gap-1 mx-auto hover:bg-emerald-100"><FileText size={14}/> Gerenciar</button>
                  </td>
                  <td className="px-6 py-4 text-sm text-center">
                    <ActionButtons onEdit={() => handleOpenForm(p)} onDelete={() => setDeleteModal({ isOpen: true, data: p })} />
                  </td>
                </tr>
              )
            })}
          </tbody>
        </table>
      </div>
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName="Processo" />
      
      {docsModal.isOpen && (
        <GerenciarDocumentosModal 
          processo={docsModal.processo}
          itemTrabalho={itensTrabalho.find(i => i.id === docsModal.processo.itemTrabalhoId)}
          gerencias={gerencias}
          onClose={() => setDocsModal({ isOpen: false, processo: null })}
          documentos={documentos}
          setDocumentos={setDocumentos}
          pessoas={pessoas}
          macroprocessos={macroprocessos}
          origensDemanda={origensDemanda}
          niveisRisco={niveisRisco}
          tiposServico={tiposServico}
        />
      )}
    </div>
  );
}

// COMPONENTE: MODAL DE DOCUMENTOS DO PROCESSO
// ==========================================
function GerenciarDocumentosModal({ processo, itemTrabalho, gerencias, onClose, documentos, setDocumentos, pessoas, macroprocessos, origensDemanda, niveisRisco, tiposServico }) {
  const [isAdding, setIsAdding] = useState(false);
  const [novoDocTipo, setNovoDocTipo] = useState('');
  
  // Controle do Editor Avançado
  const [showEditor, setShowEditor] = useState(false);
  const [editingDoc, setEditingDoc] = useState(null); // null = Criando novo, obj = Editando
  const [deleteModal, setDeleteModal] = useState({ isOpen: false, data: null });

  const processDocs = documentos.filter(d => d.processoId === processo.id).sort((a,b) => b.dataCriacao - a.dataCriacao);
  const hasOS = processDocs.some(d => d.tipo === 'Ordem de Serviço (OS)');
  const allowedTypes = hasOS ? TIPOS_DOCUMENTO : ['Ordem de Serviço (OS)'];

  // Abertura do Editor para Novo Documento
  const handleOpenEditorNew = () => {
    if (!novoDocTipo) return;
    setEditingDoc(null);
    setShowEditor(true);
  };

  // Abertura do Editor para Edição
  const handleEditDoc = (doc) => {
    setEditingDoc(doc);
    setNovoDocTipo(doc.tipo);
    setShowEditor(true);
  };

  // Callback de Save do Editor
  const handleSaveDocument = (editorData) => {
    if (editingDoc) {
      // Atualizar documento existente
      setDocumentos(documentos.map(d => 
        d.id === editingDoc.id ? { ...d, ...editorData } : d
      ));
    } else {
      // Criar novo documento
      const novoDoc = {
        id: Date.now(),
        processoId: processo.id,
        tipo: novoDocTipo,
        status: 'Minuta', 
        dataCriacao: new Date().getTime(),
        ...editorData // Spread the sections, tags, date, and customFields
      };
      setDocumentos([...documentos, novoDoc]);
    }

    setShowEditor(false);
    setIsAdding(false);
    setNovoDocTipo('');
    setEditingDoc(null);
  };

  const handleSendApproval = (docId) => {
    setDocumentos(documentos.map(d => d.id === docId ? { ...d, status: 'Aguardando Aprovação' } : d));
  };

  const executeDelete = () => {
    if (deleteModal.data) {
      setDocumentos(documentos.filter(d => d.id !== deleteModal.data.id));
      setDeleteModal({ isOpen: false, data: null });
    }
  };

  const getStatusBadge = (status) => {
    switch(status) {
      case 'Minuta': return <span className="flex items-center gap-1 bg-slate-100 text-slate-700 py-1 px-2 rounded-full text-xs font-semibold"><Clock size={12}/> Minuta</span>;
      case 'Aguardando Aprovação': return <span className="flex items-center gap-1 bg-amber-100 text-amber-700 py-1 px-2 rounded-full text-xs font-semibold"><Clock size={12}/> Pendente</span>;
      case 'Aprovado': return <span className="flex items-center gap-1 bg-emerald-100 text-emerald-700 py-1 px-2 rounded-full text-xs font-semibold"><CheckCircle2 size={12}/> Aprovado</span>;
      default: return null;
    }
  };

  return (
    <div className="fixed inset-0 bg-slate-900/60 backdrop-blur-sm z-50 flex items-center justify-center p-4">
      <div className="bg-white rounded-xl shadow-2xl w-full max-w-4xl max-h-[90vh] flex flex-col animate-in fade-in zoom-in-95 duration-200">
        
        <div className="p-6 border-b flex items-start justify-between bg-slate-50 rounded-t-xl">
          <div>
            <h2 className="text-xl font-bold text-slate-800 flex items-center gap-2">
              <FileText className="text-blue-700" /> Documentos do Processo
            </h2>
            <p className="text-sm text-slate-500 mt-1">
              Item de Trabalho: <strong className="text-slate-700">{itemTrabalho?.numeroReferencia || 'N/A'}</strong>
            </p>
          </div>
          <button onClick={onClose} className="text-slate-400 hover:text-slate-600 bg-white shadow-sm border p-2 rounded-full transition-colors"><X size={20} /></button>
        </div>

        <div className="p-6 flex-1 overflow-y-auto">
          
          <div className="flex justify-between items-center mb-6">
            <h3 className="text-lg font-bold text-slate-800">Listagem de Documentos</h3>
            {!isAdding && (
              <button 
                onClick={() => { setIsAdding(true); setNovoDocTipo(allowedTypes[0]); }}
                className="bg-blue-700 hover:bg-blue-800 text-white px-4 py-2 rounded-lg flex items-center gap-2 font-medium text-sm transition-colors"
              >
                <FilePlus size={18} /> Novo Documento
              </button>
            )}
          </div>

          {!hasOS && !isAdding && processDocs.length === 0 && (
            <div className="bg-amber-50 text-amber-800 p-4 rounded-lg mb-6 text-sm flex gap-2 border border-amber-200">
              <AlertTriangle size={20} className="shrink-0" />

                  <th className="px-6 py-3 text-sm font-semibold text-slate-600 text-center">Ações</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100 bg-white">
                {processDocs.length === 0 ? (
                  <tr>
                    <td colSpan={4} className="px-6 py-8 text-center text-slate-500 text-sm">Nenhum documento gerado para este processo ainda.</td>
                  </tr>
                ) : processDocs.map(doc => (

                    </td>
                    <td className="px-6 py-4 flex justify-center">{getStatusBadge(doc.status)}</td>
                    <td className="px-6 py-4 text-sm text-center">
                      {doc.status === 'Minuta' ? (

                          <button 
                            onClick={() => handleSendApproval(doc.id)}

                          </div>
                        </div>
                      ) : (
                        <span className="text-slate-400 text-xs italic">Ações bloqueadas</span>
                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      </div>
      
      {/* Exclusão do Documento */}
      <DeleteModal modal={deleteModal} setModal={setDeleteModal} onConfirm={executeDelete} itemName={deleteModal.data?.tipo || 'Documento'} />

      {/* Editor do Documento */}
      {showEditor && (
        <DocumentEditor 
          docType={novoDocTipo} 
          onSave={handleSaveDocument} 
          onCancel={() => { setShowEditor(false); setEditingDoc(null); }} 
          initialData={editingDoc}
          itemTrabalho={itemTrabalho}
          gerencias={gerencias}
          processo={processo}
          pessoas={pessoas}
          macroprocessos={macroprocessos}
          origensDemanda={origensDemanda}
          niveisRisco={niveisRisco}
          tiposServico={tiposServico}
        />
      )}
    </div>
  );
}

                              ))}
                              {(section.constatacoes || []).length === 0 && <div className="text-sm text-gray-500 italic p-4 bg-gray-50 text-center rounded border border-gray-200 border-dashed">Nenhuma constatação registrada nesta seção.</div>}
                            </div>
                          ) : (
                            renderTextWithElements(section.content || '', section.id)
                          )}
                      </div>
                  </section>
              )
          })}

          <div className="mt-20 flex flex-col items-center justify-center text-center break-inside-avoid">
              <div className="w-72 border-t border-gray-400 mb-3"></div>
              <p className="font-bold text-gray-900 text-sm">{renderGlobal(supervisor?.nome || 'Supervisor não definido')}</p>
              <p className="text-sm text-gray-600 font-medium">{renderGlobal('Gerente CC1')}</p>
              <p className="text-sm text-gray-600 font-medium">{renderGlobal('Gerência de Auditoria Interna (AUDIN)')}</p>
          </div>

          <div className="mt-16 pt-4 border-t border-gray-200 text-xs text-gray-500 font-medium flex justify-between items-center">
              <span>{renderGlobal(`Gerado via SisAudin em ${formatDisplayDate(docDate)}`)}</span>
          </div>
        </div>
      );
    }

    // Fallback for other document types
    return (
      <div className="w-full px-14 py-14 bg-white text-gray-800 text-sm leading-normal flex flex-col selection:bg-indigo-200 selection:text-indigo-900">
        <div className="flex justify-end mb-8"><div className="flex gap-2 text-gray-500"><span>{renderGlobal('Porto Alegre, RS')}</span><span>{renderGlobal(formatDisplayDate(docDate))}</span></div></div>
        <div className="mb-4"><strong className="block mb-1 text-base">{renderGlobal('GHC - Auditoria Interna')}</strong><p className="text-gray-500">{renderGlobal('Av. Francisco Trein, 596')}</p></div>
        <h3 className="text-center font-bold text-lg mb-6 uppercase">{renderGlobal(docType)}</h3>
        
        {itemTrabalho && ( 
          <div className="mb-8 border border-gray-300 p-4 rounded-lg bg-gray-50">
            <table className="w-full text-left">
              <tbody>
                <tr><td className="font-bold w-24 align-top py-1">{renderGlobal('Referência:')}</td><td className="align-top py-1">{renderGlobal(`${itemTrabalho.numeroReferencia}/${itemTrabalho.anoReferencia}`)}</td></tr>
                <tr><td className="font-bold align-top py-1">{renderGlobal('Período:')}</td><td className="align-top py-1">{renderGlobal(`${formatDisplayDate(customFields.dataInicial)} a ${formatDisplayDate(customFields.dataFinal)}`)}</td></tr>
                <tr><td className="font-bold align-top py-1">{renderGlobal('Objeto:')}</td><td className="align-top py-1">{renderGlobal(itemTrabalho.objetoAuditado)}</td></tr>
                <tr><td className="font-bold align-top py-1">{renderGlobal('Objetivo:')}</td><td className="align-top py-1">{renderGlobal(itemTrabalho.objetivoAuditoria)}</td></tr>
              </tbody>
            </table>
          </div>
        )}

        <div className="flex flex-col gap-6 w-full">
          {sections.map(section => (
            <div key={section.id} className="w-full mb-8">
              {section.title && <h4 className="font-bold text-base mb-3 text-gray-800 border-b pb-1 break-after-avoid">{section.title}</h4>}
              {section.type === 'informacoes-preliminares' ? (
                <div className="flex flex-col gap-4 text-gray-700">
                  {(section.items || []).map((item, idx) => (
                    <div key={item.id} className="flex gap-2 text-justify">
                      <span className="font-bold shrink-0">3.{idx + 1}</span>
                      <div className="flex-1 whitespace-pre-wrap leading-relaxed">
                        {renderTextWithElements(item.text, section.id, item.id)}
                      </div>
                    </div>
                  ))}
                </div>
              ) : section.type === 'constatacoes' ? (
                <div className="flex flex-col gap-8 text-gray-800">
                  {(section.constatacoes || []).map((constatacao, cIdx) => (
                    <div key={constatacao.id} className="flex flex-col gap-4 text-justify">
                      <h4 className="font-bold text-base border-b border-gray-200 pb-1">4.{cIdx + 1}. {constatacao.titulo || '[Título não informado]'}</h4>
                      <div className="flex flex-col gap-4">
                        <div><strong className="block text-sm mb-1 font-bold">4.{cIdx + 1}.1. Situação Encontrada</strong><div className="text-sm whitespace-pre-wrap ml-1">{renderTextWithElements(constatacao.situacao || '', section.id)}</div></div>
                        <div><strong className="block text-sm mb-1 font-bold">4.{cIdx + 1}.2. Critério</strong><div className="text-sm whitespace-pre-wrap ml-1">{renderTextWithElements(constatacao.criterio || '', section.id)}</div></div>
                        <div><strong className="block text-sm mb-1 font-bold">4.{cIdx + 1}.3. Causa</strong><div className="text-sm whitespace-pre-wrap ml-1">{renderTextWithElements(constatacao.causa || '', section.id)}</div></div>
                        <div><strong className="block text-sm mb-1 font-bold">4.{cIdx + 1}.4. Efeito (ou Risco)</strong><div className="text-sm whitespace-pre-wrap ml-1">{renderTextWithElements(constatacao.efeito || '', section.id)}</div></div>
                        <div><strong className="block text-sm mb-1 font-bold">4.{cIdx + 1}.5. Evidências</strong><div className="text-sm whitespace-pre-wrap ml-1">{renderTextWithElements(constatacao.evidencias || '', section.id)}</div></div>
                        <div className="mt-2"><strong className="block text-sm mb-2 font-bold">4.{cIdx + 1}.6. Recomendação(ões)</strong>
                          <ul className="list-decimal pl-5 flex flex-col gap-2 ml-1">
                            {(constatacao.recomendacoes || []).map((rec, rIdx) => (
                              <li key={rec.id} className="text-sm whitespace-pre-wrap"><div className="pl-1">{renderTextWithElements(rec.text || '', section.id)}</div></li>
                            ))}
                          </ul>
                        </div>
                      </div>
                    </div>
                  ))}
                </div>
              ) : (
                <div className="text-gray-700 whitespace-pre-wrap leading-relaxed text-justify" style={{ fontWeight: activeFormats.bold ? 'bold' : 'normal', fontStyle: activeFormats.italic ? 'italic' : 'normal', textDecoration: activeFormats.underline ? 'underline' : 'none', textAlign: activeFormats.align }}>
                  {renderTextWithElements(section.content || '', section.id)}
                </div>
              )}
            </div>
          ))}
        </div>
        <div className="mt-24 pb-8 break-inside-avoid"><div className="border-t border-gray-400 w-48 pt-2 mx-auto text-gray-500 text-center font-medium">Assinatura</div></div>
      </div>
    );
  };

  return (
    <div className="fixed inset-0 z-[60] bg-[#F8F9FC] flex items-center justify-center p-4 md:p-8 font-sans text-gray-800 animate-in fade-in" onClick={() => setActivePopover(null)}>
      <div className="bg-white rounded-xl shadow-lg w-full max-w-7xl h-full flex flex-col overflow-hidden border border-gray-100" onClick={(e) => e.stopPropagation()}>
        
        {/* Header */}
        <div className="px-6 py-5 border-b border-gray-100 flex justify-between items-center bg-white z-10 flex-wrap gap-4">
          <h1 className="text-xl font-bold text-gray-900">{initialData ? 'Editar Documento:' : 'Criar Novo Documento:'} <span className="text-blue-600">{docType}</span></h1>
          
          <div className="flex items-center gap-4">
            
            <div className="flex items-center gap-2 bg-slate-50 p-1 rounded-lg border border-slate-200">
              <button onClick={toggleSupervisionMode} className={`flex items-center gap-2 text-sm px-3 py-1.5 rounded-md transition-colors ${isSupervisionMode ? 'bg-indigo-100 text-indigo-800 font-bold' : 'text-gray-500 hover:text-gray-800'}`}>
                {isSupervisionMode ? <UserCheck size={16} /> : <User size={16} />} Supervisão
              </button>
              
              {isSupervisionMode && (
                <div className="flex items-center gap-1 animate-in fade-in slide-in-from-left-2 border-l border-slate-200 pl-2">
                  <button 
                    onMouseDown={(e) => { e.preventDefault(); if(selectionData) openAnnotationModal('comentario'); }} 
                    disabled={!selectionData} 
                    className={`flex items-center gap-1 text-xs px-3 py-1.5 rounded transition-all ${selectionData ? 'bg-indigo-600 text-white hover:bg-indigo-700 shadow-sm cursor-pointer' : 'bg-slate-200 text-slate-400 cursor-not-allowed opacity-60'}`} 
                    title="Inserir Comentário"
                  >
                    <MessageSquare size={14} /> Comentar
                  </button>
                  <button 
                    onMouseDown={(e) => { e.preventDefault(); if(selectionData && !selectionData.isGlobal) openAnnotationModal('sugestao'); }} 
                    disabled={!selectionData || selectionData.isGlobal} 
                    className={`flex items-center gap-1 text-xs px-3 py-1.5 rounded transition-all ${selectionData && !selectionData.isGlobal ? 'bg-amber-500 text-white hover:bg-amber-600 shadow-sm cursor-pointer' : 'bg-slate-200 text-slate-400 cursor-not-allowed opacity-60'}`} 
                    title={selectionData?.isGlobal ? "Não é possível sugerir alterações em áreas geradas automaticamente" : "Sugerir Alteração"}
                  >
                    <PenTool size={14} /> Sugerir
                </button>
              </div>
            )}
          </div>

          <div className="w-px h-5 bg-gray-200"></div>
          
          <button onClick={() => {
            const nextEdit = !showEditPanel;
            setShowEditPanel(nextEdit);
            if (!nextEdit) setShowPreview(true);
          }} className="flex items-center gap-2 text-sm text-gray-500 hover:text-gray-800 bg-gray-50 hover:bg-gray-100 px-3 py-1.5 rounded-lg border border-gray-200 transition-colors">
            {showEditPanel ? <><EyeOff size={16} /> Ocultar edição</> : <><Edit2 size={16} /> Mostrar edição</>}
          </button>

          <button onClick={() => {
            const nextPreview = !showPreview;
            setShowPreview(nextPreview);
            if (!nextPreview) setShowEditPanel(true);
          }} className="flex items-center gap-2 text-sm text-gray-500 hover:text-gray-800 bg-gray-50 hover:bg-gray-100 px-3 py-1.5 rounded-lg border border-gray-200 transition-colors">
            {showPreview ? <><EyeOff size={16} /> Ocultar preview</> : <><Eye size={16} /> Mostrar preview</>}
          </button>
          
          <div className="w-px h-5 bg-gray-200"></div>
          
          <button onClick={onCancel} className="text-gray-400 hover:text-gray-600 transition-colors"><X size={24} strokeWidth={2} /></button>
        </div>
      </div>

      {/* Content */}
      <div className="flex flex-col lg:flex-row flex-1 overflow-hidden">
        {/* Editor Column */}
        {showEditPanel && (
        <div className={`p-6 md:p-8 flex flex-col gap-6 overflow-y-auto bg-white transition-all ${showPreview ? 'w-full lg:w-[55%]' : 'w-full'}`} onClick={() => setActivePopover(null)}>
          
          {isSupervisionMode && (
            <div className="bg-indigo-50 border border-indigo-100 rounded-lg p-4 text-sm text-indigo-800 flex items-start gap-3 shadow-inner">
                <UserCheck size={20} className="text-indigo-600 mt-0.5 shrink-0" />
                <span className="leading-relaxed">
                  O <strong>Modo Supervisão</strong> está ativo. <br/>
                  Para inserir anotações, <strong>selecione um trecho de texto na área de pré-visualização (à direita)</strong> e use os botões <strong>"Comentar"</strong> ou <strong>"Sugerir"</strong> localizados no cabeçalho.
                </span>
              </div>
            )}

            <div className="flex flex-col sm:flex-row gap-4">
              <div className="flex flex-col gap-2 w-full sm:w-[200px]">
                <label className="text-sm font-semibold text-gray-700">Data do Documento</label>
                <div className="relative">
                  <input type="date" value={docDate} onChange={(e) => setDocDate(e.target.value)} className="block w-full pl-3 pr-10 py-2.5 border-gray-200 border rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500 text-gray-600 bg-transparent z-10" />
                </div>
              </div>
              <div className="flex flex-col gap-2 w-full sm:w-[200px]">
                <label className="text-sm font-semibold text-gray-700">Data Inicial *</label>
                <div className="relative">
                  <input type="date" value={customFields.dataInicial} onChange={(e) => setCustomFields({...customFields, dataInicial: e.target.value})} className="block w-full pl-3 pr-10 py-2.5 border-blue-200 border rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500 text-blue-800 bg-blue-50 z-10" />
                </div>
              </div>
              <div className="flex flex-col gap-2 w-full sm:w-[200px]">
                <label className="text-sm font-semibold text-gray-700">Data Final *</label>
                <div className="relative">
                  <input type="date" value={customFields.dataFinal} onChange={(e) => setCustomFields({...customFields, dataFinal: e.target.value})} className="block w-full pl-3 pr-10 py-2.5 border-blue-200 border rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500 text-blue-800 bg-blue-50 z-10" />
                </div>
              </div>
            </div>

            <div className="flex flex-wrap gap-2">
              {tags.map((tag, index) => (
                <div key={index} className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-slate-100 text-xs text-gray-600">
                  {tag} <button onClick={() => removeTag(index)} className="text-gray-400 hover:text-gray-800"><X size={12} /></button>
                </div>
              ))}
            </div>

            {/* ---- INÍCIO DOS CAMPOS PERSONALIZADOS POR TIPO DE DOC ---- */}
            
            {/* Informações do Item de Trabalho e Processo (Read-Only) */}
            {itemTrabalho && (
              <div className="flex flex-col gap-4">
                <div className="bg-blue-50 border border-blue-100 rounded-lg p-4">
                  <h4 className="text-xs font-bold text-blue-800 uppercase tracking-wider mb-3 flex items-center gap-1"><Info size={14}/> Informações do Item de Trabalho</h4>
                  <div className="grid grid-cols-1 md:grid-cols-3 gap-3 text-sm text-blue-900">
                    <div><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Ano</span> {itemTrabalho.anoReferencia}</div>
                    <div><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Nº Referência</span> {itemTrabalho.numeroReferencia}</div>
                    <div><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Nível de Risco</span> {risco?.nome || '-'}</div>
                    
                    <div><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Macroprocesso</span> {macro?.nome || '-'}</div>
                    <div className="md:col-span-2"><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Origem da Demanda</span> {origem?.nome || '-'}</div>

                    <div className="md:col-span-3"><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Objeto</span> {itemTrabalho.objetoAuditado}</div>
                    <div className="md:col-span-3"><span className="font-semibold block text-blue-700 text-xs uppercase mb-0.5">Objetivo</span> {itemTrabalho.objetivoAuditoria}</div>
                  </div>
                </div>

                {processo && (
                  <div className="bg-indigo-50 border border-indigo-100 rounded-lg p-4">
                    <h4 className="text-xs font-bold text-indigo-800 uppercase tracking-wider mb-3 flex items-center gap-1"><Users size={14}/> Equipe do Processo</h4>
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-3 text-sm text-indigo-900">
                      <div><span className="font-semibold block text-indigo-700 text-xs uppercase mb-0.5">Supervisor</span> {supervisor?.nome || 'Não definido'}</div>
                      <div><span className="font-semibold block text-indigo-700 text-xs uppercase mb-0.5">Responsável</span> {responsavel?.nome || 'Não definido'}</div>
                      <div className="md:col-span-2">
                        <span className="font-semibold block text-indigo-700 text-xs uppercase mb-0.5">Membros da Equipe</span>
                        <div className="flex flex-wrap gap-2 mt-1">
                          {equipe.length > 0 ? equipe.map(p => (
                            <span key={p.id} className="bg-white border border-indigo-200 px-2 py-1 rounded text-xs">{p.nome}</span>
                          )) : <span className="text-indigo-400 italic">Nenhum membro vinculado</span>}
                        </div>
                      </div>
                    </div>
                  </div>
                )}
              </div>
            )}

            {/* Campos Específicos: Solicitação de Auditoria */}
            {docType === 'Solicitação de Auditoria' && (
              <div className="bg-slate-50 border border-slate-200 rounded-lg p-4 mt-2 space-y-4">
                <h4 className="text-xs font-bold text-slate-700 uppercase tracking-wider flex items-center gap-1"><List size={14}/> Configurações da Solicitação</h4>
                <div className="relative z-20">
                  <MultiSearchableSelect
                    label="Gerências Destinatárias"
                    options={gerencias ? gerencias.map(g => ({ id: g.id, label: `${g.sigla} - ${g.nome}` })) : []}
                    values={customFields.gerenciasDestinatarias}
                    onChange={(vals) => setCustomFields({...customFields, gerenciasDestinatarias: vals})}
                    placeholder="Selecione as gerências"
                  />
                </div>
                <div>
                  <label className="block text-sm font-medium text-slate-700 mb-2">Data Limite de Atendimento</label>
                  <input
                    type="date"
                    value={customFields.dataLimite}
                    onChange={(e) => setCustomFields({...customFields, dataLimite: e.target.value})}
                    className="w-full p-2.5 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
                  />
                </div>
              </div>
            )}
            
            {/* ---- FIM DOS CAMPOS PERSONALIZADOS ---- */}

            <div className="flex flex-col gap-6 mt-2">
              {sections.map((section) => (
                <div key={section.id} className="flex flex-col gap-2">
                  <div className="flex justify-between items-center">
                    <input value={section.title} onChange={(e) => updateSectionTitle(section.id, e.target.value)} className="text-sm font-semibold text-gray-700 bg-transparent border-none focus:ring-1 focus:ring-blue-500 rounded px-1 py-0.5 w-full max-w-[70%]" placeholder="Título da seção" />
                    {!section.isDefault && <button onClick={() => removeSection(section.id)} className="text-red-500 text-xs flex items-center gap-1 hover:bg-red-50 px-2 py-1 rounded"><X size={14} /> Remover</button>}
                  </div>
                  
                  <div className={`border border-gray-200 rounded-lg overflow-hidden flex flex-col transition-opacity ${isSupervisionMode ? 'opacity-70 grayscale-[20%]' : ''}`}>
                    {section.type !== 'informacoes-preliminares' && section.type !== 'constatacoes' && (
                      <div className="flex items-center gap-1 p-2 border-b border-gray-100 bg-slate-50 flex-wrap">
                        <button disabled={isSupervisionMode} onClick={() => toggleFormat('bold')} className={`p-1.5 rounded disabled:cursor-not-allowed ${activeFormats.bold ? 'bg-blue-100 text-blue-600' : 'text-gray-600 hover:bg-gray-200'}`}><Bold size={16} /></button>
                        <button disabled={isSupervisionMode} onClick={() => toggleFormat('italic')} className={`p-1.5 rounded disabled:cursor-not-allowed ${activeFormats.italic ? 'bg-blue-100 text-blue-600' : 'text-gray-600 hover:bg-gray-200'}`}><Italic size={16} /></button>
                        <button disabled={isSupervisionMode} onClick={() => toggleFormat('underline')} className={`p-1.5 rounded disabled:cursor-not-allowed ${activeFormats.underline ? 'bg-blue-100 text-blue-600' : 'text-gray-600 hover:bg-gray-200'}`}><Underline size={16} /></button>
                        <div className="w-px h-5 bg-gray-200 mx-1"></div>
                        <button disabled={isSupervisionMode} onClick={() => setAlign('left')} className={`p-1.5 rounded disabled:cursor-not-allowed ${activeFormats.align === 'left' ? 'bg-blue-100 text-blue-600' : 'text-gray-600 hover:bg-gray-200'}`}><AlignLeft size={16} /></button>
                        <button disabled={isSupervisionMode} onClick={() => setAlign('center')} className={`p-1.5 rounded disabled:cursor-not-allowed ${activeFormats.align === 'center' ? 'bg-blue-100 text-blue-600' : 'text-gray-600 hover:bg-gray-200'}`}><AlignCenter size={16} /></button>
                        <button disabled={isSupervisionMode} onClick={() => setAlign('right')} className={`p-1.5 rounded disabled:cursor-not-allowed ${activeFormats.align === 'right' ? 'bg-blue-100 text-blue-600' : 'text-gray-600 hover:bg-gray-200'}`}><AlignRight size={16} /></button>
                        <div className="w-px h-5 bg-gray-200 mx-1"></div>
                        <button disabled={isSupervisionMode} onClick={() => { setActiveSectionId(section.id); setShowImageModal(true); }} className="p-1.5 text-gray-600 hover:bg-gray-200 disabled:cursor-not-allowed rounded"><ImageIcon size={16} /></button>
                        <button disabled={isSupervisionMode} onClick={() => { setActiveSectionId(section.id); setShowTableModal(true); }} className="p-1.5 text-gray-600 hover:bg-gray-200 disabled:cursor-not-allowed rounded"><TableIcon size={16} /></button>
                      </div>
                    )}
                    
                    {section.type === 'informacoes-preliminares' ? (
                      <div className="flex flex-col gap-3 p-4 bg-slate-50 min-h-[160px]">
                        <p className="text-xs text-slate-500 font-medium flex items-center gap-1.5 mb-1"><Info size={14}/> Arraste os blocos pelo ícone pontilhado para alterar a ordem da numeração (3.1, 3.2).</p>
                        {(section.items || []).map((item, idx) => (
                          <div 
                            key={item.id}
                            draggable={!isSupervisionMode}
                            onDragStart={(e) => {
                              e.dataTransfer.effectAllowed = "move";
                              setDraggedItemInfo({ index: idx, sectionId: section.id });
                            }}
                            onDragOver={(e) => {
                              e.preventDefault();
                              e.dataTransfer.dropEffect = "move";
                            }}
                            onDrop={(e) => {
                              e.preventDefault();
                              if (draggedItemInfo && draggedItemInfo.sectionId === section.id && draggedItemInfo.index !== idx) {
                                setSections(prev => prev.map(s => {
                                  if (s.id === section.id) {
                                    const newItems = [...s.items];
                                    const [dragged] = newItems.splice(draggedItemInfo.index, 1);
                                    newItems.splice(idx, 0, dragged);
                                    return { ...s, items: newItems };
                                  }
                                  return s;
                                }));
                              }
                              setDraggedItemInfo(null);
                            }}
                            className={`flex items-start gap-3 bg-white border border-gray-200 p-3 rounded-lg shadow-sm transition-all ${draggedItemInfo?.index === idx ? 'opacity-50 border-blue-400 scale-[0.99]' : ''}`}
                          >
                            <div className={`mt-2 ${isSupervisionMode ? 'text-gray-300 cursor-not-allowed' : 'cursor-grab active:cursor-grabbing text-gray-400 hover:text-blue-500'}`} title="Arraste para reordenar">
                              <GripVertical size={20} />
                            </div>
                            <div className="mt-2 font-bold text-gray-500 text-sm shrink-0 w-8">
                              3.{idx + 1}
                            </div>
                            <textarea 
                              value={item.text}
                              onChange={(e) => {
                                setSections(prev => prev.map(s => s.id === section.id ? { 
                                  ...s, items: s.items.map((it, i) => i === idx ? { ...it, text: e.target.value } : it) 
                                } : s));
                              }}
                              onFocus={() => setActiveSectionId(section.id)}
                              disabled={isSupervisionMode}
                              className="flex-1 min-h-[80px] p-2 text-sm border border-gray-100 hover:border-gray-200 focus:outline-none focus:ring-1 focus:ring-blue-500 focus:border-blue-500 rounded bg-transparent resize-y disabled:cursor-not-allowed"
                              placeholder="Descreva a informação preliminar..."
                            />
                            <button 
                              onClick={() => setDeletePrelimModal({ isOpen: true, sectionId: section.id, itemIndex: idx })}
                              disabled={isSupervisionMode}
                              className="mt-1 p-2 text-red-500 hover:bg-red-50 hover:text-red-700 rounded transition-colors disabled:opacity-50 disabled:cursor-not-allowed"
                              title="Excluir informação"
                            >
                              <Trash2 size={18} />
                            </button>
                          </div>
                        ))}
                        <button 
                          onClick={() => {
                            setSections(prev => prev.map(s => s.id === section.id ? { 
                              ...s, items: [...(s.items || []), { id: Date.now() + Math.random(), text: '' }] 
                            } : s));

                          className="mt-2 flex items-center justify-center gap-2 py-3 border border-dashed border-gray-300 rounded text-blue-600 hover:bg-blue-50 text-sm font-medium transition-colors disabled:opacity-50 disabled:cursor-not-allowed bg-white"
                        >
                          <Plus size={18} /> Adicionar Informação Preliminar
                        </button>
                      </div>
                    ) : section.type === 'constatacoes' ? (
                      <div className="flex flex-col gap-6 p-4 bg-slate-50 min-h-[160px] rounded-b-lg">

                              if (draggedConstInfo && draggedConstInfo.sectionId === section.id && draggedConstInfo.index !== cIdx) {

                                  if (s.id === section.id) {

                                    const [dragged] = newConst.splice(draggedConstInfo.index, 1);

                            <div className="p-4 flex flex-col gap-5">
                               <div>
                                  <label className="text-[10px] font-bold text-gray-400 uppercase tracking-wider">Título da Constatação *</label>

                               </div>
                               
                               <div className="grid grid-cols-1 gap-4">
                                 {[
                                   {key: 'situacao', label: 'Situação Encontrada (Condição)'},
                                   {key: 'criterio', label: 'Critério'},
                                   {key: 'causa', label: 'Causa'},
                                   {key: 'efeito', label: 'Efeito (ou Risco)'},

                                        onDragOver={(e) => { e.preventDefault(); e.stopPropagation(); e.dataTransfer.dropEffect = "move"; }}
                                        onDrop={(e) => {

                                          if (draggedRecInfo && draggedRecInfo.sectionId === section.id && draggedRecInfo.constIndex === cIdx && draggedRecInfo.recIndex !== rIdx) {
                                            setSections(prev => prev.map(s => {

                                                const newConst = [...s.constatacoes];
                                                const newRecs = [...newConst[cIdx].recomendacoes];
                                                const [dragged] = newRecs.splice(draggedRecInfo.recIndex, 1);
                                                newRecs.splice(rIdx, 0, dragged);
                                                newConst[cIdx] = { ...newConst[cIdx], recomendacoes: newRecs };

                                        className={`flex items-start gap-3 bg-white border border-emerald-100 p-3 rounded-lg shadow-sm transition-all ${draggedRecInfo?.recIndex === rIdx && draggedRecInfo?.constIndex === cIdx ? 'opacity-50 border-emerald-400 scale-[0.99]' : ''}`}
                                      >
                                         <div className={`mt-2 ${isSupervisionMode ? 'text-gray-300' : 'cursor-grab active:cursor-grabbing text-gray-400 hover:text-emerald-600'}`} title="Arraste para reordenar a recomendação"><GripVertical size={16}/></div>
                                         <div className="mt-2 font-bold text-emerald-700 text-xs shrink-0 w-5 h-5 flex items-center justify-center bg-emerald-50 rounded-full border border-emerald-200">{rIdx + 1}</div>
                                         <textarea value={rec.text || ''} onChange={(e) => {
                                            setSections(prev => prev.map(s => {
                                              if (s.id === section.id) {

                                                const newRecs = [...newConst[cIdx].recomendacoes];
                                                newRecs[rIdx] = { ...newRecs[rIdx], text: e.target.value };

                                              return s;

                                         <button disabled={isSupervisionMode} onClick={() => setDeleteRecModal({ isOpen: true, sectionId: section.id, constIndex: cIdx, recIndex: rIdx })} className="mt-1 p-2 text-red-400 hover:text-red-600 hover:bg-red-50 rounded disabled:opacity-50 transition-colors" title="Excluir Recomendação"><Trash2 size={16}/></button>

                                   {(constatacao.recomendacoes || []).length === 0 && <div className="text-xs text-emerald-600/70 italic text-center py-5 bg-emerald-50/40 rounded-lg border border-emerald-200/50 border-dashed">Nenhuma recomendação inserida nesta constatação.</div>}

                               </div>

                          </div>

                        <button disabled={isSupervisionMode} onClick={() => {

                            ...s, constatacoes: [...(s.constatacoes || []), { id: Date.now() + Math.random(), titulo: '', situacao: '', criterio: '', causa: '', efeito: '', evidencias: '', recomendacoes: [] }]
                          } : s));

                    ) : (
                      <textarea 
                        disabled={isSupervisionMode}
                        className="w-full h-40 p-4 resize-none focus:outline-none text-gray-700 text-sm leading-relaxed disabled:bg-slate-50 disabled:cursor-not-allowed"
                        style={{ fontWeight: activeFormats.bold ? 'bold' : 'normal', fontStyle: activeFormats.italic ? 'italic' : 'normal', textDecoration: activeFormats.underline ? 'underline' : 'none', textAlign: activeFormats.align }}
                        value={section.content}

                  </div>
                </div>
              ))}
              <button disabled={isSupervisionMode} onClick={addSection} className="flex items-center justify-center gap-2 py-3 border-2 border-dashed border-gray-300 rounded-lg text-gray-500 hover:text-blue-600 hover:bg-blue-50 disabled:opacity-50 disabled:cursor-not-allowed transition-colors text-sm">
              <PlusCircle size={18} /> Adicionar seção

          </div>

        )}

        {/* Preview Column */}
        {showPreview && (
          <div className={`bg-[#F4F6F9] border-gray-100 p-6 flex flex-col relative min-h-[500px] transition-all ${showEditPanel ? 'w-full lg:w-[45%] border-l' : 'w-full'}`}>
            <div className="flex justify-between items-center mb-6 border-b border-gray-200 pb-4">
              <h2 className="text-sm font-semibold text-gray-600 flex items-center gap-2">
                Pré-visualização {isSupervisionMode && <span className="bg-indigo-100 text-indigo-700 text-[10px] px-2 py-0.5 rounded uppercase font-bold tracking-wider">Modo Seleção</span>}
              </h2>
                
                <div className="flex items-center gap-3">
                  <span className="text-xs text-gray-400 hidden 2xl:inline-block" title="Estimativa de páginas ao imprimir">~{estimatedPages} pág(s)</span>
                  
                  {/* Controles de Zoom Fixos no Topo */}

                  <button onClick={handlePrint} className="flex items-center gap-1.5 text-xs font-semibold px-3 py-1.5 bg-white border border-gray-200 text-gray-700 hover:bg-gray-50 hover:text-blue-600 rounded-md shadow-sm transition-colors">

              <div 
                className={`flex-1 flex flex-col items-center overflow-y-auto overflow-x-hidden relative ${isSupervisionMode ? 'cursor-text' : ''}`} 
                onScroll={() => setActivePopover(null)}
                onMouseUp={handlePreviewSelect}
                onKeyUp={handlePreviewSelect}
              >
                <div className="absolute opacity-0 pointer-events-none left-[-9999px]"><div ref={contentRef}>{renderDocumentContent()}</div></div>
                
                {/* Visualização Contínua - Evita o corte artificial de textos do DOM */}
                <div className="flex flex-col gap-6 origin-top transition-transform pb-16 pt-2" style={{ transform: `scale(${zoom / 100})` }}>
                  <div className="bg-white rounded shadow-md relative shrink-0 min-h-[1123px]" style={{ width: PAGE_WIDTH }}>
                     {renderDocumentContent()}

                </div>
              </div>
            </div>

        </div>

        <div className="px-6 py-4 border-t border-gray-100 flex justify-end gap-3 bg-white">
          <button onClick={onCancel} className="px-6 py-2.5 border border-gray-300 text-gray-700 font-medium rounded-lg hover:bg-gray-50 text-sm transition-colors">Cancelar</button>

      {/* Modal de Anotação (Supervisão) */}
      {annotationModal.isOpen && (
        <div className="fixed inset-0 bg-gray-900/50 z-[70] flex items-center justify-center p-4 backdrop-blur-sm">
          <div className="bg-white rounded-xl shadow-xl w-full max-w-md flex flex-col overflow-hidden animate-in fade-in zoom-in-95 duration-200" onClick={(e) => e.stopPropagation()}>
            <div className="px-6 py-4 border-b border-gray-100 flex justify-between items-center">
              <h2 className="text-lg font-bold text-gray-900 flex items-center gap-2">
                {annotationModal.type === 'comentario' ? <><MessageSquare size={18} className="text-indigo-600"/> Inserir Comentário</> : <><PenTool size={18} className="text-amber-500"/> Inserir Sugestão de Texto</>}

                <X size={20} strokeWidth={2} />
              </button>
            </div>
            
            <div className="p-6 flex flex-col gap-4">
              <div className="bg-gray-50 p-3 rounded-lg border border-gray-100 text-xs text-gray-600">
                <strong className="block mb-2 text-gray-800">Trecho selecionado na pré-visualização:</strong>
                <div className="bg-white p-2.5 border border-gray-200 rounded max-h-32 overflow-y-auto whitespace-pre-wrap italic text-gray-700">
                  {selectionData?.text}

              </div>

              <div className="flex flex-col gap-1.5">
                <label className="text-sm font-semibold text-gray-700">

                </label>
                <textarea 
                  value={annotationInput}
                  onChange={e => setAnnotationInput(e.target.value)}
                  className="w-full h-32 px-3 py-2 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500 text-sm text-gray-700 shadow-sm"
                  placeholder={annotationModal.type === 'comentario' ? 'Digite aqui o comentário...' : 'Edite o texto para sugerir a alteração...'}
                  autoFocus
                />
              </div>
            </div>

            <div className="px-6 py-4 border-t border-gray-100 flex justify-end gap-3 bg-gray-50">
              <button 
                onClick={closeAnnotationModal}
                className="px-4 py-2 text-gray-700 font-medium hover:bg-gray-200 rounded-lg transition-colors text-sm"
              >
                Cancelar
              </button>
              <button

                disabled={!annotationInput.trim()}
                className={`px-4 py-2 text-white font-medium rounded-lg transition-colors text-sm ${annotationModal.type === 'comentario' ? 'bg-indigo-600 hover:bg-indigo-700' : 'bg-amber-500 hover:bg-amber-600'} disabled:opacity-50 disabled:cursor-not-allowed`}
              >
                Salvar Anotação
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Popover Fixo de visualização das anotações existentes */}
      {activePopover && (
        <div 
          className="fixed z-[80] bg-white border border-gray-200 shadow-2xl rounded-lg text-left pointer-events-auto flex flex-col gap-2 p-3 animate-in fade-in zoom-in-95 duration-100"
          style={{
            top: activePopover.rect.top - 8,
            left: activePopover.rect.left + (activePopover.rect.width / 2),
            transform: 'translate(-50%, -100%)',
            width: activePopover.type.includes('comentario') ? '288px' : '224px'
          }}
          onClick={(e) => e.stopPropagation()}
        >
          <span className="absolute top-full left-1/2 -translate-x-1/2 border-4 border-transparent border-t-white drop-shadow-sm"></span>
          
          {activePopover.type === 'global-comentario' ? (
            <>
              <strong className="block text-[10px] text-red-600 font-bold uppercase tracking-wider">Comentário (Área Fixa/Automática)</strong>

                <label className="text-[9px] font-semibold text-gray-500">Comentário do supervisor:</label>
                <textarea 
                  value={activePopover.content}
                  readOnly
                  className="w-full h-16 p-2 text-xs bg-gray-50 border border-gray-100 rounded resize-none text-gray-600 outline-none"
                />
              </div>
              
              <div className="flex gap-2 mt-2">
                <button 
                  onClick={() => handleGlobalCiente(activePopover.commentId)}
                  className="w-full bg-gray-500 hover:bg-gray-600 text-white text-[10px] py-1.5 rounded font-bold transition-colors shadow-sm"
                >
                  Ciente

          ) : activePopover.type === 'comentario' ? (
            <>
              <strong className="block text-[10px] text-red-600 font-bold uppercase tracking-wider">Comentário</strong>

                <label className="text-[9px] font-semibold text-gray-500">Comentário do supervisor:</label>

                  className="w-full h-16 p-2 text-xs bg-gray-50 border border-gray-100 rounded resize-none text-gray-600 outline-none"

                <button

                >
                  Concordar e Alterar

                  className="flex-1 bg-blue-500 hover:bg-blue-600 text-white text-[10px] py-1.5 rounded font-bold transition-colors shadow-sm"
                >
                  Manter
                </button>
              </div>
            </>
          ) : (
            <>
              <strong className="block mb-1 text-[10px] text-amber-600 font-bold uppercase tracking-wider">Sugestão</strong>
              <span className="block mb-3 text-xs text-gray-800 leading-snug bg-gray-50 p-2 rounded border border-gray-100 italic">
                "{activePopover.content}"
              </span>
              
              <div className="flex gap-2">
                <button 
                  onClick={() => handleSuggestionAction(activePopover.sectionId, activePopover.itemId, activePopover.rawPart, 'aceitar', activePopover.content, activePopover.annotatedText, activePopover.user, activePopover.date)}
                  className="flex-1 bg-green-500 hover:bg-green-600 text-white text-[10px] py-1.5 rounded font-bold transition-colors shadow-sm"
                >
                  Aceitar
                </button>
                <button 
                  onClick={() => handleSuggestionAction(activePopover.sectionId, activePopover.itemId, activePopover.rawPart, 'manter', activePopover.content, activePopover.annotatedText, activePopover.user, activePopover.date)}
                  className="flex-1 bg-blue-500 hover:bg-blue-600 text-white text-[10px] py-1.5 rounded font-bold transition-colors shadow-sm"
                >
                  Manter
                </button>
              </div>
            </>
          )}
        </div>
      )}

      {showImageModal && (
        <div className="fixed inset-0 bg-gray-900/50 z-[70] flex items-center justify-center p-4 backdrop-blur-sm">
          <div className="bg-white rounded-xl shadow-xl w-full max-w-md flex flex-col overflow-hidden animate-in fade-in zoom-in-95 duration-200" onClick={(e) => e.stopPropagation()}>
            <div className="px-6 py-4 border-b border-gray-100 flex justify-between items-center">
              <h2 className="text-lg font-bold text-gray-900">Inserir Imagem</h2>

                <X size={20} strokeWidth={2} />
              </button>
            </div>
            
            <div className="p-6 flex flex-col gap-4">
              <div className="flex flex-col gap-1.5">
                <label className="text-sm font-semibold text-gray-700">Título da imagem *</label>

                  value={newImage.title}
                  onChange={e => setNewImage({...newImage, title: e.target.value})}

                />
              </div>

              <div className="flex flex-col gap-1.5">
                <label className="text-sm font-semibold text-gray-700">Data de obtenção</label>
                <input 
                  type="date" 
                  value={newImage.date}
                  onChange={e => setNewImage({...newImage, date: e.target.value})}
                  className="w-full px-3 py-2 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500 text-gray-600"

              </div>
              <div className="flex flex-col gap-1.5">
                <label className="text-sm font-semibold text-gray-700">Fonte</label>
                <input 
                  type="text" 
                  value={newImage.source}
                  onChange={e => setNewImage({...newImage, source: e.target.value})}
                  className="w-full px-3 py-2 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500"
                />
              </div>
            </div>

            <div className="px-6 py-4 border-t border-gray-100 flex justify-end gap-3 bg-gray-50">
              <button 
                onClick={() => setShowImageModal(false)}
                className="px-4 py-2 text-gray-700 font-medium hover:bg-gray-200 rounded-lg transition-colors text-sm"
              >
                Cancelar
              </button>
              <button 
                onClick={handleAddImage}
                className="px-4 py-2 bg-blue-500 hover:bg-blue-600 text-white font-medium rounded-lg transition-colors text-sm"
              >
                Inserir
              </button>
            </div>
          </div>
        </div>
      )}

      {showTableModal && (
        <div className="fixed inset-0 bg-gray-900/50 z-[70] flex items-center justify-center p-4 backdrop-blur-sm">
          <div className="bg-white rounded-xl shadow-xl w-full max-w-4xl max-h-[90vh] flex flex-col overflow-hidden animate-in fade-in zoom-in-95 duration-200" onClick={(e) => e.stopPropagation()}>
            <div className="px-6 py-4 border-b border-gray-100 flex justify-between items-center">
              <h2 className="text-lg font-bold text-gray-900">Construtor de Tabela</h2>
              <button onClick={() => setShowTableModal(false)} className="text-gray-400 hover:text-gray-600 transition-colors">
                <X size={20} strokeWidth={2} />
              </button>
            </div>
            
            <div className="p-6 flex flex-col gap-6 overflow-y-auto">
              {/* Configuração Inicial */}
              <div className="flex flex-col sm:flex-row gap-4 items-end bg-gray-50 p-4 rounded-lg border border-gray-200">
                <div className="flex flex-col gap-1.5 flex-1">
                  <label className="text-sm font-semibold text-gray-700">Título da Tabela *</label>
                  <input 
                    type="text" 
                    value={tbConfig.title}
                    onChange={e => setTbConfig({...tbConfig, title: e.target.value})}
                    className="w-full px-3 py-2 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500"
                    placeholder="Ex: Orçamento"
                  />
                </div>
                <div className="flex flex-col gap-1.5 w-24">
                  <label className="text-sm font-semibold text-gray-700">Linhas</label>
                  <input 
                    type="number" min="1" max="20"
                    value={tbConfig.rows}
                    onChange={e => setTbConfig({...tbConfig, rows: parseInt(e.target.value) || 1})}
                    className="w-full px-3 py-2 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500"
                  />
                </div>
                <div className="flex flex-col gap-1.5 w-24">
                  <label className="text-sm font-semibold text-gray-700">Colunas</label>
                  <input 
                    type="number" min="1" max="15"
                    value={tbConfig.cols}
                    onChange={e => setTbConfig({...tbConfig, cols: parseInt(e.target.value) || 1})}
                    className="w-full px-3 py-2 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-500"
                  />
                </div>
                <button 
                  onClick={generateTableGrid}
                  className="px-4 py-2 bg-gray-800 hover:bg-gray-900 text-white font-medium rounded-lg transition-colors h-[42px]"
                >
                  Gerar Grelha
                </button>
              </div>

              {/* Área do Construtor de Grelha */}
              {tbConfig.grid.length > 0 && (
                <div className="flex flex-col gap-3 border border-gray-200 rounded-lg p-4">
                  <div className="flex justify-between items-center pb-3 border-b border-gray-100">
                    <button 
                      onClick={() => setTbConfig({...tbConfig, selectionMode: !tbConfig.selectionMode, selectedCells: []})}
                      className={`flex items-center gap-2 px-3 py-1.5 rounded-md text-sm font-medium transition-colors border ${tbConfig.selectionMode ? 'bg-blue-50 border-blue-200 text-blue-700' : 'bg-white border-gray-200 text-gray-600 hover:bg-gray-50'}`}
                    >
                      <MousePointerClick size={16} />
                      {tbConfig.selectionMode ? 'Desativar Modo de Seleção' : 'Modo de Seleção para Concatenar'}
                    </button>
                    
                    <button 
                      onClick={handleMergeCells}
                      disabled={tbConfig.selectedCells.length < 2}
                      className="flex items-center gap-2 px-3 py-1.5 bg-amber-500 hover:bg-amber-600 disabled:bg-gray-300 disabled:cursor-not-allowed text-white rounded-md text-sm font-medium transition-colors"
                    >
                      <Link size={16} /> Concatenar Células
                    </button>
                  </div>

                  {tbConfig.selectionMode && (
                    <div className="text-xs text-blue-600 bg-blue-50 p-2 rounded">
                      Clique nas células que deseja selecionar para concatenar. O texto das células concatenadas será preservado.
                    </div>
                  )}

                  <div className="overflow-x-auto py-2">
                    <table className="w-full border-collapse border border-gray-300 table-fixed min-w-full">
                      <tbody>
                        {tbConfig.grid.map((row, rIdx) => (
                          <tr key={rIdx}>
                            {row.map((cell, cIdx) => {
                              if (cell.isMerged) return null;
                              const isSelected = tbConfig.selectedCells.some(c => c.r === rIdx && c.c === cIdx);
                              
                              return (
                                <td
                                  key={cIdx}
                                  rowSpan={cell.rowSpan}
                                  colSpan={cell.colSpan}
                                  onClick={() => toggleTableCellSelection(rIdx, cIdx)}
                                  className={`border p-1.5 align-top transition-colors 
                                    ${tbConfig.selectionMode ? 'cursor-pointer hover:bg-blue-50/50' : ''} 
                                    ${isSelected ? 'bg-blue-100 border-blue-500 shadow-inner' : 'border-gray-300'}`}
                                >
                                  <textarea
                                    value={cell.text}
                                    readOnly={tbConfig.selectionMode}
                                    onChange={(e) => handleCellTextChange(rIdx, cIdx, e.target.value)}
                                    className={`w-full min-h-[40px] h-full resize-none p-1 text-sm bg-transparent outline-none ${tbConfig.selectionMode ? 'pointer-events-none cursor-pointer' : 'focus:bg-gray-50'}`}
                                    placeholder={tbConfig.selectionMode ? '' : "Texto..."}
                                  />
                                </td>
                              );
                            })}
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>
              )}
            </div>

            <div className="px-6 py-4 border-t border-gray-100 flex justify-end gap-3 bg-gray-50 mt-auto">
              <button 
                onClick={() => setShowTableModal(false)}
                className="px-4 py-2 text-gray-700 font-medium hover:bg-gray-200 rounded-lg transition-colors text-sm"
              >
                Cancelar
              </button>
              <button 
                onClick={handleInsertTable}
                className="px-4 py-2 bg-blue-500 hover:bg-blue-600 text-white font-medium rounded-lg transition-colors text-sm"
              >
                Inserir Tabela
              </button>
            </div>
          </div>
        </div>
      )}

      <DeleteModal 
        modal={deletePrelimModal} 
        setModal={setDeletePrelimModal} 
        onConfirm={executeDeletePrelim} 
        itemName="esta Informação Preliminar" 
      />
      <DeleteModal 
        modal={deleteConstModal} 
        setModal={setDeleteConstModal} 
        onConfirm={executeDeleteConst} 
        itemName="esta Constatação Inteira e todas as suas recomendações" 
      />
      <DeleteModal 
        modal={deleteRecModal} 
        setModal={setDeleteRecModal} 
        onConfirm={executeDeleteRec} 
        itemName="esta Recomendação Específica" 
      />
    </div>
  );
}
