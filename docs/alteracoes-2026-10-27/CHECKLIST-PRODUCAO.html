6. Nginx e front

Atualizar o bloco server do nginx.conf

Substituir só o bloco server { … } ativo. Acrescenta: anexos até 50 MB (hoje 1 MB), /uploads para o backend, tempo maior para PDF, cabeçalhos X-Forwarded-*, index.html sem cache e compressão.

	server {
		listen 80;
		server_name localhost;

		client_max_body_size 50m;

		gzip       on;
		gzip_types text/css application/javascript application/json image/svg+xml;

		location / {
			root   html;
			index  index.html index.htm;
			try_files $uri $uri/ /index.html;
		}

		location = /index.html {
			root   html;
			add_header Cache-Control "no-cache";
		}

		location ~ ^/(api|uploads)/ {
			proxy_pass http://127.0.0.1:8080;
			proxy_http_version 1.1;
			proxy_set_header Upgrade $http_upgrade;
			proxy_set_header Connection 'upgrade';
			proxy_set_header Host $host;
			proxy_set_header X-Real-IP $remote_addr;
			proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
			proxy_set_header X-Forwarded-Proto $scheme;
			proxy_cache_bypass $http_upgrade;
			proxy_read_timeout 120s;
		}
	}
